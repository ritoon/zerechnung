  1. {{.T.start_install}}

    Shell
    go get github.com/ritoon/zerechnung

  2. {{.T.start_create}}

    Go
    pdf, err := zerechnung.Create(host, inv, &zerechnung.CreateOptions{
	Calculate: true,                // totals and VAT breakdown
	Validate:  &validate.Options{}, // refuse an invalid invoice
})

  3. {{.T.start_validate}}

    Go
    report, _, err := zerechnung.Check(pdf, &validate.Options{Auto: true})
if err != nil {
	log.Fatal(err)
}
for _, f := range report.Errors() {
	fmt.Println(f)
}

  4. {{.T.start_cli}}

    Shell
    go install github.com/ritoon/zerechnung/cmd/zerechnung@latest
zerechnung validate -auto invoice.pdf
